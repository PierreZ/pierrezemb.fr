{# ========================================================================== macros.html — reusable Tera helpers ========================================================================== #} {# Recursive table-of-contents list. Renders page.toc and its nested children (h2 → h3 → h4 …) as nested
    s, hierarchically numbered (1, 1.1, 1.2 …), capped at `depth` levels deep. #} {% macro toc_list(items, prefix, depth) %}
      {%- for h in items %} {%- set num = prefix ~ loop.index %}
    1. {{ num }}{{ h.title }} {%- if depth > 1 and h.children %}{{ self::toc_list(items=h.children, prefix=num ~ ".", depth=depth - 1) }}{% endif %}
      2. {%- endfor %}
    {% endmacro toc_list %} {# A single nomenclature (BOM) row for a post listing. #} {% macro bom_row(page) %}
    {{ page.date | date(format="%Y-%m-%d") }}~{{ page.reading_time }} min
    {{ page.title }} {%- if page.description %}{{ page.description }}{% endif %}
    {% endmacro bom_row %} {%- macro github_markdown_source(relative_path) -%} {%- set revision = get_env(name="ZOLA_SOURCE_REVISION", default="master") -%} {%- set source = "https://raw.githubusercontent.com/PierreZ/pierrezemb.fr/" ~ revision ~ "/content/" ~ relative_path -%} {{ source | safe }} {%- endmacro github_markdown_source -%}
